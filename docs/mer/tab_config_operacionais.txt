CREATE TABLE config_operacionais (
	id INTEGER PRIMARY KEY CHECK(id = 1),
	data_criacao TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
	criado_por INTEGER NOT NULL,
	data_atualizacao TEXT, 
	atualizado_por INTEGER,

	/*configurações de venda*/
	operacao TEXT NOT NULL DEFAULT 'VENDAS_E_ORCAMENTOS' CHECK(operacao IN ('SOMENTE_ORCAMENTOS', 'SOMENTE_VENDAS', 'VENDAS_E_ORCAMENTOS')),
	controle_caixa INTEGER NOT NULL DEFAULT 1 CHECK(controle_caixa IN (0, 1)),
	
	/*configurações para envio de e-mail automática*/
	email_principal TEXT,
	smtp TEXT NOT NULL DEFAULT 'smtp.gmail.com',
	porta INTEGER NOT NULL DEFAULT 587 CHECK(porta > 0),
	protocolo_seguranca TEXT NOT NULL DEFAULT 'TLS' CHECK(protocolo_seguranca IN ('TLS', 'SSL')),
	senha_app TEXT,
	
	/*outras configurações*/
	path_logo TEXT,
	path_impressora TEXT,

	FOREIGN KEY(criado_por)
	REFERENCES usuarios(id),

	FOREIGN KEY(atualizado_por)
	REFERENCES usuarios(id)
)
